    build(            \
        "26/10/2026", \

        "octjan"                \
    )

    exit
}

function build(start,end,assigned,leave,prefix)
{
    run("gawk -v start=\"" start "\" -v end=\"" end \
        "\" -f pivot2.txt " assigned \
        " > " prefix "_pivot.csv")

    run("gawk -f roster_to_SICmzn.txt " \
        prefix "_pivot.csv > " \
        prefix "_roster.mzn")

    run("gawk -v start=\"" start "\" -v end=\"" end \
        "\" -f leave_to_mzn.txt " leave \
        " > " prefix "_leave.mzn")
}

function run(cmd)
{
    print cmd

    if (system(cmd))
    {
        print "FAILED: " cmd
        exit 1
    }

}
